        "giftNumber": gift_id,
        "page": 1,
        "limit": 20

        data = await response.json()
        result = extract_gift(data, gift_id)

        }

MODEL_NAME_MAP = {
    'eternalrose': 'EternalRose',
    'eternal rose': 'EternalRose',
    'eternal_rose': 'EternalRose',
    'evileye': 'EvilEye',
    'evil eye': 'EvilEye',
    ...
}

def normalize_model_name(text: str) -> str:
    text = text.strip().lower()

    # Проверка в маппинге
    if text in MODEL_NAME_MAP:
        return MODEL_NAME_MAP[text]

    # Капитализация каждого слова
    words = text.split()
    return ''.join(word.capitalize() for word in words)

Почему 7 секунд:
Peek API имеет rate limit
7 секунд = безопасный интервал
Проверка остановки каждую секунду
Retry логика в API
max_retries = 3
for attempt in range(max_retries):
    try:
        response = await session.get(PEEK_URL, params=params)

        if response.status == 429:  # Rate limit
            wait_time = 3 + (attempt * 2)  # 3, 5, 7 секунд
            await asyncio.sleep(wait_time)
            continue

        return data
    except Exception as e:
        if attempt < max_retries - 1:
            await asyncio.sleep(2)
        else:
            raise

Многосообщенный вывод (Split Messages)
Проблема: Telegram лимит 4096 символов
Решение:
def format_gift_summary(gifts, model, start, end) -> list[str]:
    messages = []
    current_msg = "..."

    for gift in gifts:
        gift_line = f"{i}. #{num} | @{owner}\n"

        # Проверка лимита (оставляем запас)
        if len(current_msg + gift_line) > 3800:
            messages.append(current_msg)
            current_msg = f"Продолжение ({len(messages) + 1})\n\n" + gift_line
        else:
            current_msg += gift_line

    messages.append(current_msg)
    return messages

🐛 ИЗВЕСТНЫЕ ПРОБЛЕМЫ И РЕШЕНИЯ
1. База данных сбрасывается на Railway
Проблема: Ephemeral storage - файл удаляется при рестарте
Решение (будущее):
Использовать PostgreSQL addon
Или внешний сервис (Supabase, MongoDB Atlas)
2. API может меняться
Проблема: Peek API неофициальный, может измениться
Решение:
Вся работа с API в peek_api.py
Легко заменить на другой источник
3. Rate Limiting
Проблема: При быстрых запросах API блокирует
Решение:
Задержка 7 секунд между запросами
Retry логика (3 попытки)
Увеличение задержки при 429 ошибке
